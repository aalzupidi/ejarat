        function updateElectricityCalcField(id, field, value) {
            const item = electricityCalcBills.find(i => i.id === id);
            if (!item) return;
            item[field] = parseFloat(value) || 0;

            const curr = parseFloat(item.currentReading) || 0;
            const prev = parseFloat(item.prevReading) || 0;
            const qty = curr - prev;
            const consVal = parseFloat((qty > 0 ? qty * 0.18 : 0).toFixed(2));
            item.consumptionVal = consVal;
            const serv = parseFloat(item.meterService) || 0;
            const arrears = parseFloat(item.arrears) || 0;
            const vatRate = item.vat !== undefined ? item.vat : 15;
            const consWithTax = consVal * (1 + vatRate / 100);
            item.value = parseFloat((consWithTax + serv + arrears).toFixed(2));

            renderUI();
            saveDataLocally(true);
        }

        function updateWaterCalcField(id, field, value) {
            const item = waterCalcBills.find(i => i.id === id);
            if (!item) return;
            item[field] = parseFloat(value) || 0;

            const curr = parseFloat(item.currentValue) || 0;
            const prev = parseFloat(item.previousValue) || 0;
            const ton = parseFloat(item.pricePerTon) || 0;
            const arrears = parseFloat(item.arrears) || 0;
            const qty = curr - prev;
            const subTotal = parseFloat(((qty > 0 ? qty : 0) * ton).toFixed(2));
            const vatRate = item.vat !== undefined ? item.vat : 15;
            item.value = parseFloat((subTotal * (1 + vatRate / 100) + arrears).toFixed(2));

            renderUI();
            saveDataLocally(true);
        }

        function toWhatsAppNumber(phone) {
            let d = String(phone || '').replace(/\D/g, '');
            if (!d) return '';
            if (d.startsWith('00')) d = d.slice(2);
            if (d.startsWith('05') && d.length === 10) d = '966' + d.slice(1);
            else if (d.startsWith('5') && d.length === 9) d = '966' + d;
            return d;
        }

        function handleAuthSubmit() {
            const email = document.getElementById('authEmail').value.trim();
            const pass = document.getElementById('authPassword').value.trim();
            const submitBtn = document.getElementById('authSubmitBtn');
            const mode = submitBtn.dataset.mode || 'login';

            if (!email || !pass) {
                alert('الرجاء إدخال البريد الإلكتروني وكلمة المرور');
                return;
            }
            if (!initFirebase()) {
                alert('يرجى ضبط إعدادات Firebase أولاً من القائمة الجانبية.');
                openFirebaseConfigModal();
                return;
            }

            if (mode === 'login') {
                auth.signInWithEmailAndPassword(email, pass).then(res => {
                    currentUser = res.user;
                    updateAuthModalUI();
                    closeAuthModal();
                    showToast('تم تسجيل الدخول بنجاح');
                    loadCloudData();
                }).catch(err => alert('خطأ في تسجيل الدخول: ' + err.message));
            } else {
                auth.createUserWithEmailAndPassword(email, pass).then(res => {
                    currentUser = res.user;
                    updateAuthModalUI();
                    closeAuthModal();
                    showToast('تم إنشاء الحساب وتسجيل الدخول بنجاح');
                    saveDataLocally(false);
                }).catch(err => alert('خطأ في إنشاء الحساب: ' + err.message));
            }
        }

        function loadCloudData() {
            if (!db || !currentUser) return;
            if (unsubscribeSnapshot) unsubscribeSnapshot();
            unsubscribeSnapshot = db.collection('users').doc(currentUser.uid).onSnapshot(doc => {
                if (doc.exists) {
                    const data = doc.data();
                    isSyncingFromCloud = true;
                    if (data.tenants) tenants = data.tenants;
                    if (data.electricityCalcBills) electricityCalcBills = data.electricityCalcBills;
                    if (data.waterCalcBills) waterCalcBills = data.waterCalcBills;
                    if (data.electricityCalcHistory) electricityCalcHistory = data.electricityCalcHistory;
                    if (data.waterCalcHistory) waterCalcHistory = data.waterCalcHistory;
                    renderUI();
                    isSyncingFromCloud = false;
                }
            });
        }
